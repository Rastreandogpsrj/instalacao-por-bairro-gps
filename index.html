async function loadBairrosData() { const response = await fetch("bairros.json"); if (!response.ok) throw new Error("Erro ao carregar dados de bairros"); return await response.json(); } async function verValor() { const bairroInput = document.getElementById("bairroInput"); const quantidadeInput = document.getElementById("quantidadeInput"); const tipoRastreador = document.getElementById("tipoRastreador"); const consultarBtn = document.getElementById("consultarBtn"); // Reset UI document.getElementById("resultBox").style.display = "none"; document.getElementById("mensalidadeBox").style.display = "none"; document.getElementById("beneficiosBox").style.display = "none"; document.getElementById("ctaButtons").style.display = "none"; try { if (!bairroInput.value.trim()) throw new Error("Por favor, digite o nome do bairro"); const quantidade = parseInt(quantidadeInput.value); if (isNaN(quantidade) || quantidade < 1) throw new Error("Quantidade inválida"); consultarBtn.disabled = true; consultarBtn.textContent = "Calculando..."; const data = await loadBairrosData(); const searchTerm = bairroInput.value.trim() .normalize("NFD").replace(/[\u0300-\u036f]/g, "") .toLowerCase(); const resultado = data.find(entry => entry.bairro.normalize("NFD").replace(/[\u0300-\u036f]/g, "") .toLowerCase().includes(searchTerm) ); const resultBox = document.getElementById("resultBox"); const mensalidadeBox = document.getElementById("mensalidadeBox"); const beneficiosBox = document.getElementById("beneficiosBox"); if (resultado) { const valorInstalacao = parseFloat(resultado.valor) * quantidade; resultBox.className = "result-box success"; resultBox.innerHTML = `
R$ ${valorInstalacao.toFixed(2)}

Valor total para instalação em ${resultado.bairro}

`; if (tipoRastreador.value === "com") { mensalidadeBox.className = "result-box success"; mensalidadeBox.innerHTML = `

📝 Plano com Bloqueio

R$ 49,90/mês

(Valor promocional para pagamento até o vencimento)

De: R$ 54,90

`; beneficiosBox.className = "result-box success"; beneficiosBox.innerHTML = `

✨ Benefícios Inclusos:

✓ Localização em tempo real

✓ Bloqueio/desbloqueio remoto

✓ Histórico de rotas detalhado

✓ Alertas de ignição e movimento

✓ Suporte 24/7 especializado

`; } else { mensalidadeBox.className = "result-box success"; mensalidadeBox.innerHTML = `

📝 Plano Básico

R$ 39,90/mês
`; beneficiosBox.className = "result-box success"; beneficiosBox.innerHTML = `

✨ Benefícios Inclusos:

✓ Localização em tempo real

✓ Histórico de rotas básico

✓ Alertas de ignição

✓ Suporte especializado

`; } document.getElementById("ctaButtons").style.display = "block"; } else { resultBox.className = "result-box error"; resultBox.innerHTML = `

❌ Bairro não encontrado

Verifique a ortografia ou entre em contato para verificar cobertura.

`; } resultBox.style.display = "block"; mensalidadeBox.style.display = "block"; beneficiosBox.style.display = "block"; } catch (error) { const resultBox = document.getElementById("resultBox"); resultBox.className = "result-box error"; resultBox.innerHTML = `

⚠️ ${error.message}

Tente novamente ou entre em contato conosco.

`; resultBox.style.display = "block"; console.error("Erro:", error); } finally { consultarBtn.disabled = false; consultarBtn.textContent = "Calcular Valor"; } } // Enter tecla ativa o botão document.getElementById("bairroInput").addEventListener("keypress", (e) => { if (e.key === "Enter") verValor(); });
